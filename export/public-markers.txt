# Publishable markers: generic shapes of citations of the project's working records, and of
# links into the paths that stay private.
#
# Each line is ID<TAB>REGEX, a Python regular expression. tools/export_lists.py --check refuses
# a pattern that does not compile or can match an empty string, and a public exception whose
# line no longer holds a hit of the marker it names. These shapes need no private input, so a
# check of the public repository can run them too.

# Backlog and record identifiers.
id-row	\b(?:[Bb]acklog |TODO )?[Rr]ows? [0-9]+\.[0-9]+\b
id-todo	\bTODO(?:\.md)? (?:under |row )?[0-9]+(?:\.[0-9]+)?\b
id-done	\bDONE [0-9]+\.[0-9]+\b
id-d-number	\bO?D-[0-9]+\b
id-f-name	\bF-[A-Z][A-Z0-9]*(?:-[A-Z0-9]+)+\b

# Links into development history, links into the paths that stay private, and any repository
# named as a private one.
link-history	github\.com/[A-Za-z0-9-]+/secureconfig/(?:pull|commit|compare|actions)/
link-denied-path	github\.com/[A-Za-z0-9-]+/secureconfig/(?:blob|tree|raw)/[^/\s]+/(?:(?:AGENTS|CLAUDE|DECISIONS|DONE|PENDING-DECISIONS|TODO)\.md|\.aiqt|\.baseline|\.github|scripts|site-src|tools|export/FORMAT\.md|export/private-paths\.txt|site/draft|(?:[A-Za-z0-9._+@-]+/)*[A-Za-z0-9._+@-]+-latest\.zip)(?![A-Za-z0-9._+@-])
link-private-repo	github\.com/[A-Za-z0-9-]+/[A-Za-z0-9._-]+_private\b
