# Publishable markers: generic shapes of citations of the project's working records, which no
# published file may hold.
#
# Each line is ID<TAB>REGEX, a Python regular expression searched in every line of every
# published file. A hit fails the export unless public-exceptions.tsv lists that exact line.
# These shapes need no private input, so a check of the public repository can run them too.

# Backlog and record identifiers.
id-row	\b(?:[Bb]acklog |TODO )?[Rr]ows? [0-9]+\.[0-9]+\b
id-todo	\bTODO(?:\.md)? (?:under |row )?[0-9]+(?:\.[0-9]+)?\b
id-done	\bDONE [0-9]+\.[0-9]+\b
id-d-number	\bO?D-[0-9]+\b
id-f-name	\bF-[A-Z][A-Z0-9]*(?:-[A-Z0-9]+)+\b

# Links into development history, and any repository named as a private one.
link-history	github\.com/[A-Za-z0-9-]+/secureconfig/(?:pull|commit|compare|actions)/
link-private-repo	github\.com/[A-Za-z0-9-]+/[A-Za-z0-9._-]+_private\b
